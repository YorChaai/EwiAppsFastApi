  Future<void> _handleNotificationTap(String rawPath) async {
    if (!mounted) return;
    final path = rawPath.trim().toLowerCase();

    // Extract ID from path
    final parts = path.split('/');
    int? id;
    for (int i = parts.length - 1; i >= 0; i--) {
      final value = int.tryParse(parts[i]);
      if (value != null) {
        id = value;
        break;
      }
    }

    // Navigate based on path
    if (path.contains('/settlements') && id != null) {
      // Navigate to settlement detail
      Navigator.pushReplacement(
        context,
        MaterialPageRoute(
          builder: (_) => SettlementDetailScreen(settlementId: id!),
        ),
      );
      return;
    }

    if (path.contains('/advances') && id != null) {
      // Navigate to advance detail
      Navigator.pushReplacement(
        context,
        MaterialPageRoute(builder: (_) => AdvanceDetailScreen(advanceId: id!)),
      );
      return;
    }

    if (path.contains('/categories')) {
      // Navigate to dashboard categories tab
      if (context.read<AuthProvider>().isManager) {
        Navigator.of(context).pushNamedAndRemoveUntil(
          '/',
          (route) => false,
          arguments: {'initialTabIndex': 3},
        );
      }
      return;
    }

    if (path.contains('/reports')) {
      // Navigate to dashboard reports tab
      if (context.read<AuthProvider>().isManager) {
        Navigator.of(context).pushNamedAndRemoveUntil(
          '/',
          (route) => false,
          arguments: {'initialTabIndex': 2},
        );
      }
      return;
    }
  }

  Color _statusColor(String status) {
    switch (status) {
      case 'draft':
        return AppTheme.textSecondary;
      case 'submitted':
        return AppTheme.warning;
      case 'approved':
        return AppTheme.success;
      case 'revision_draft':
        return AppTheme.accent;
      case 'revision_submitted':
        return AppTheme.warning;
      case 'revision_rejected':
        return AppTheme.danger;
      case 'in_settlement':
      case 'rejected':
      case 'completed':
      case 'settled':
        return status == 'rejected' ? AppTheme.danger : AppTheme.primary;
      default:
        return AppTheme.textSecondary;
    }
  }

  Color _advanceStatusColor(Map<String, dynamic> advance) {
    final status = (advance['status'] ?? 'draft').toString().toLowerCase();
    if (status == 'in_settlement') {
      final settlementStatus =
          (advance['settlement_status'] ?? '').toString().toLowerCase();
      final isSettlementApproved =
          settlementStatus == 'approved' || settlementStatus == 'completed';
      return isSettlementApproved ? AppTheme.success : AppTheme.danger;
    }
    return _statusColor(status);
  }

  List<Map<String, dynamic>> _approvedAdvanceItems(
    Map<String, dynamic> advance,
  ) {
    final items = _asMapList(advance['items']);
    return items.where((item) {
      final status = (item['status'] ?? '').toString().toLowerCase();
      return status == 'approved' || status == 'completed';
    }).toList();
  }

  String _advanceFundSource(Map<String, dynamic> item) {
    for (final key in const [
      'source_bank_name',
      'source_bank',
      'bank_name',
      'bank',
      'source',
    ]) {
      final value = (item[key] ?? '').toString().trim();
      if (value.isNotEmpty) return value;
    }
    return '-';
  }

  String _advanceDisbursementSourceLabel(Map<String, dynamic> advance) {
    final saved = (advance['disbursement_source'] ?? '').toString().trim();
    if (saved.isNotEmpty && saved.toLowerCase() != 'null') return saved;
    final approvedItems = _approvedAdvanceItems(advance);
    if (approvedItems.isEmpty) return '-';
    final firstSource = _advanceFundSource(approvedItems.first);
    final allSame =
        approvedItems.every((item) => _advanceFundSource(item) == firstSource);
    return allSame &&
            firstSource.isNotEmpty &&
            firstSource.toLowerCase() != 'null'
        ? firstSource
        : '-';
  }

  String _advanceReturnSourceLabel(Map<String, dynamic> advance) {
    final returnSource = (advance['return_source'] ?? '').toString().trim();
    if (returnSource.isNotEmpty && returnSource.toLowerCase() != 'null')
      return returnSource;
    final disbursementSource = _advanceDisbursementSourceLabel(advance);
    return disbursementSource.isNotEmpty &&
            disbursementSource.toLowerCase() != 'null'
        ? disbursementSource
        : '-';
  }

  List<String> _advanceFundSourceOptions(Map<String, dynamic> advance) {
    final options = <String>{};

    void collect(dynamic raw) {
      if (raw is List) {
        for (final item in raw) {
          if (item == null) continue;
          if (item is Map) {
            final map = Map<String, dynamic>.from(item);
            for (final key in const ['name', 'bank_name', 'label', 'code']) {
              final value = (map[key] ?? '').toString().trim();
              if (value.isNotEmpty &&
                  value.toLowerCase() != 'null' &&
                  value != '-') {
                options.add(value);
                break;
              }
            }
          } else {
            final value = item.toString().trim();
            if (value.isNotEmpty &&
                value.toLowerCase() != 'null' &&
                value != '-') {
              options.add(value);
            }
          }
        }
      }
    }

    collect(advance['fund_sources']);
    collect([advance['disbursement_source'], advance['return_source']]);

    for (final item in _approvedAdvanceItems(advance)) {
      final source = _advanceFundSource(item);
      if (source.isNotEmpty &&
          source.toLowerCase() != 'null' &&
          source != '-') {
        options.add(source);
      }
    }

    final merged = <String>['-', ..._advanceFundSourceChoices];
    for (final option in options) {
      final value = option.trim();
      if (value.isNotEmpty &&
          value != '-' &&
          !_advanceFundSourceChoices.contains(value) &&
          !merged.contains(value)) {
        merged.add(value);
      }
    }
    return merged;
  }

  Widget _buildAdvanceFundReturnSection(
    BuildContext context,
    Map<String, dynamic> advance,
    AuthProvider auth,
  ) {
    // GATING: Tabel 3 dikunci jika Tabel 2 belum di-Approve Manager
    final mainStatus = (advance['status'] ?? 'draft').toString().toLowerCase();
    final isDisbursementApproved =
        (advance['is_disbursement_approved'] ?? false) ||
            mainStatus == 'completed' ||
            mainStatus == 'approved';

    if (!isDisbursementApproved) {
      return const SizedBox.shrink();
    }

    final dbReturned = _toDouble(advance['returned_amount']);
    if (dbReturned <= 0) {
      return const SizedBox.shrink();
    }

    final isManager = auth.isManager;
    final isReturnApproved = (advance['is_return_approved'] ?? false) ||
        mainStatus == 'completed' ||
        mainStatus == 'approved';
    final returnStatus =
        (advance['return_status'] ?? 'pending').toString().toLowerCase();

    // Permission logic
    final isOwner = auth.user?['id'] == advance['user_id'];
    final isInDraft = advance['status'] == 'draft' ||
        advance['status'] == 'rejected' ||
        advance['status'] == 'revision_draft' ||
        advance['status'] == 'revision_rejected';
    final isSubmittedDoc = advance['status'] == 'submitted' ||
        advance['status'] == 'revision_submitted';

    final canEditReturn = (isOwner || isManager) && isInDraft;

    // Permission logic for Table 3 Actions (Manager Only)
    final canApproveReturn =
        isManager && isSubmittedDoc && returnStatus == 'submitted';
    final canRejectReturn = isManager &&

  String _displayAdvanceStatus(Map<String, dynamic>? a) {
    if (a == null) return 'draft';
    final status = (a['status'] ?? 'draft').toString().toLowerCase();
    final returnStatus = (a['return_status'] ?? '').toString().toLowerCase();
    final items = _asMapList(a['items']);
    final hasRejectedItem = items
        .any((e) => (e['status'] ?? '').toString().toLowerCase() == 'rejected');

    if (status == 'submitted' &&
        (returnStatus == 'rejected' || hasRejectedItem)) {
      return 'revision';
    }
    return status == 'completed' ? 'approved' : status;
  }

  Future<void> _approveReturn() async {
    if (_isSaving) return;

    final confirmed = await AppDialogs.showConfirmDialog(
      context,
      title: 'Setujui Pengembalian',
      message: 'Apakah Anda yakin ingin menyetujui pengembalian dana ini?',
      confirmLabel: 'Ya, Setujui',
      confirmColor: AppTheme.success,
      icon: Icons.check_circle_outline_rounded,
    );

    if (!confirmed) return;
    if (!mounted) return;

    setState(() => _isSaving = true);
    final prov = context.read<AdvanceProvider>();
    final success = await prov.approveReturn(widget.advanceId);
    if (mounted) {
      setState(() => _isSaving = false);
      _showGlobalSnackBar(
        success
            ? 'Tabel Pengembalian disetujui ✓'
            : (prov.error ?? 'Gagal approve'),
        success ? AppTheme.success : AppTheme.danger,
      );
      if (success) {
        prov.loadAdvance(widget.advanceId);
      }
    }
  }

  Future<void> _showRejectReturnDialog() async {
    final prov = context.read<AdvanceProvider>();
    final adv = prov.currentAdvance;
    if (adv == null) return;

    List<Map<String, dynamic>> existingReasons = [];
    try {
      final decoded = jsonDecode(adv['return_notes'] ?? '[]');
      if (decoded is List) {
        existingReasons = List<Map<String, dynamic>>.from(decoded);
      }
    } catch (_) {}

    final controllers = [TextEditingController()];
    List<Map<String, dynamic>> reasonsList = [];

    final confirmReject = await showDialog<bool>(
      context: context,
      builder: (ctx) => StatefulBuilder(
        builder: (ctx, setDialogState) {
          return AlertDialog(
            backgroundColor: _cardColor(ctx),
            shape: RoundedRectangleBorder(
              borderRadius: BorderRadius.circular(20),
            ),
            title: Text(
              'Reject Pengembalian Dana',
              style: TextStyle(color: _creamColor(ctx)),
            ),
            content: SizedBox(
              width: 500,
              child: Column(
                mainAxisSize: MainAxisSize.min,
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    'Berikan alasan penolakan dalam bentuk checklist:',
                    style: TextStyle(
                      color: _secondaryText(ctx),
                      fontSize: 13,
                    ),
                  ),
                  const SizedBox(height: 16),
                  ConstrainedBox(
                    constraints: const BoxConstraints(maxHeight: 300),
                    child: SingleChildScrollView(
                      child: Column(
                        children: controllers.asMap().entries.map((entry) {
                          final idx = entry.key;
                          final ctrl = entry.value;
                          return Padding(
                            key: ValueKey(ctrl),
                            padding: const EdgeInsets.only(bottom: 8),
                            child: Row(
                              children: [
                                Icon(
                                  Icons.check_box_outline_blank_rounded,
                                  size: 20,
                                  color: _secondaryText(ctx),
                                ),
                                const SizedBox(width: 8),
                                Expanded(
                                  child: TextField(
                                    controller: ctrl,
                                    autofocus: idx == controllers.length - 1,
                                    decoration: InputDecoration(
                                      hintText: 'Alasan #${idx + 1}',
                                      hintStyle: TextStyle(
                                        color: _secondaryText(ctx)
                                            .withValues(alpha: 0.5),
                                        fontSize: 13,
                                      ),
                                      isDense: true,
                                      contentPadding:
                                          const EdgeInsets.symmetric(
                                        vertical: 8,
                                        horizontal: 10,
                                      ),
                                      border: OutlineInputBorder(
                                        borderRadius: BorderRadius.circular(8),
                                      ),
                                    ),
                                    style: TextStyle(
                                      color: _creamColor(ctx),
                                      fontSize: 13,
                                    ),
                                    maxLines: null,
                                    minLines: 1,
                                    keyboardType: TextInputType.multiline,
                                  ),
                                ),
                                if (controllers.length > 1)
                                  IconButton(
                                    icon: const Icon(
                                      Icons.remove_circle_outline_rounded,
                                      color: AppTheme.danger,
                                      size: 20,
                                    ),
                                    onPressed: () {
                                      setDialogState(() {
                                        ctrl.dispose();
                                        controllers.removeAt(idx);
                                      });
                                    },
                                  ),
                              ],
                            ),
                          );
                        }).toList(),
                      ),
                    ),
                  ),
                  const SizedBox(height: 12),
                  TextButton.icon(
                    onPressed: () {
                      setDialogState(() {
                        controllers.add(TextEditingController());
                      });
                    },
                    icon: const Icon(Icons.add_circle_outline_rounded,
                        size: 18, color: AppTheme.primary),
                    label: const Text('Tambah Alasan'),
                    style: TextButton.styleFrom(
                      foregroundColor: AppTheme.primary,
                      padding: EdgeInsets.zero,
                    ),
                  ),
                ],
              ),
            ),
            actions: [
              TextButton(
                onPressed: () => Navigator.pop(ctx, false),
                child: const Text('Batal'),
              ),
              ElevatedButton(
                onPressed: () async {
                  if (controllers.every((c) => c.text.trim().isEmpty)) return;

                  final reallyConfirm = await AppDialogs.showConfirmDialog(
                    ctx,
                    title: 'Konfirmasi Reject',
                    message:
                        'Apakah Anda yakin ingin me-reject pengembalian ini?',
                    confirmLabel: 'Ya, Reject',
                    confirmColor: AppTheme.danger,
                    icon: Icons.error_outline_rounded,
                  );

                  if (reallyConfirm == true) {
                    reasonsList = controllers
                        .where((c) => c.text.trim().isNotEmpty)
                        .map((c) => {'text': c.text.trim(), 'checked': false})
                        .toList();
                    if (ctx.mounted) Navigator.pop(ctx, true);
                  }
                },
                style: ElevatedButton.styleFrom(
                  backgroundColor: AppTheme.danger,
                ),
                child: const Text('Reject Pengembalian'),
              ),
            ],
          );
        },
      ),
    );

    if (confirmReject == true && mounted) {
      final List<Map<String, dynamic>> finalReasons = [
        ...existingReasons,
        ...reasonsList
      ];
      final notes = jsonEncode(finalReasons);

      setState(() => _isSaving = true);
      final success = await prov.rejectReturn(widget.advanceId, notes);
      if (mounted) {
        setState(() => _isSaving = false);
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text(success
                ? 'Pengembalian dana ditolak ✓'
                : (prov.error ?? 'Gagal reject')),
            backgroundColor: success ? AppTheme.success : AppTheme.danger,
          ),
        );
      }
    }

    // Cleanup
    for (var c in controllers) {
      try {
        c.dispose();
      } catch (_) {}
    }
  }

  List<Widget> _buildAdvanceDetailActions(
    BuildContext context,
    Map<String, dynamic> adv,
    AuthProvider auth,
    List<Map<String, dynamic>> workflowItems,
    bool canEditHeader,
    bool canSubmit,
    bool canCreateSettlement,
    bool canStartRevision,
    bool canShowDownloadButtons,
  ) {
    final actions = <Widget>[];

    final items = _asMapList(adv['items']);
    if (items.isNotEmpty) {
      actions.add(
        SettlementActionButton(
          onPressed: () => _showAllItemsPreview(items, adv),
          icon: Icons.list_alt_rounded,
          label: 'Pratinjau List',
          isOutlined: true,
          color: AppTheme.accent,
        ),
      );
    }

    if (canEditHeader) {
      actions.add(
        SettlementActionButton(
          onPressed: () => _showEditAdvanceDialog(context, adv),
          icon: Icons.edit_rounded,
          label: 'Edit',
          isOutlined: true,
        ),
      );
    }

    if (canSubmit) {
      final label = adv['status'] == 'revision_draft' ||
              adv['status'] == 'revision_rejected'
          ? 'Submit Revisi'
          : 'Submit';
      actions.add(
        SettlementActionButton(
          onPressed: _canSubmitAdvanceItems(workflowItems, adv)
              ? () => _submitAdvance()
              : null,
          icon: Icons.send_rounded,
          label: label,
        ),
      );
    }

    final isOwner = auth.user?['id'] == adv['user_id'];
    final isManager = auth.isManager;

    if ((isOwner || isManager) &&
        (adv['status'] == 'submitted' ||
            adv['status'] == 'rejected' ||
            adv['status'] == 'revision_submitted' ||
            adv['status'] == 'revision_rejected' ||
            adv['status'] == 'approved' ||
            adv['status'] == 'completed')) {
      actions.add(
        SettlementActionButton(
          onPressed: _moveToDraft,
          icon: Icons.undo_rounded,
          label: 'Move to Draft',
          isOutlined: true,
          color: AppTheme.warning,
        ),
      );
    }

    if (_isManager &&
        (adv['status'] == 'submitted' ||
            adv['status'] == 'revision_submitted')) {
      final label = adv['status'] == 'revision_submitted'
          ? 'Approved Final Revisi ${adv['active_revision_no'] ?? ''}'
          : 'Approved Final';

      final isDisbursementApproved = adv['is_disbursement_approved'] ?? false;

      // Gating logic for showing the button
      final itemsApproved = _canApproveAdvanceItems(workflowItems);
      // Logic for showing & clickability: Require Table 1 and Table 2 only.
      final shouldShowFinalApprove = itemsApproved && isDisbursementApproved;
      final canFinalApproveGating = shouldShowFinalApprove;

      if (shouldShowFinalApprove) {
        actions.add(
          SettlementActionButton(
            onPressed: canFinalApproveGating ? () => _approveAdvance() : null,
            icon: Icons.check_circle_outline,
            label: label,
            color: AppTheme.success,
          ),
        );
      }
    }

    if (canStartRevision) {
      final label = 'Tambah Revisi ${(adv['approved_revision_no'] ?? 0) + 1}';
      actions.add(
        SettlementActionButton(
          onPressed: () => _startRevision(),
          icon: Icons.add_circle_outline,
          label: label,
          color: AppTheme.accent,
        ),
      );
    }

    if (canShowDownloadButtons) {
      actions.add(
        SettlementActionButton(
          onPressed: () => _downloadReceipt(),
          icon: Icons.picture_as_pdf_rounded,
          label: 'PDF',
          isOutlined: true,
        ),
      );
      actions.add(
        SettlementActionButton(
          onPressed: () => _exportAdvanceExcel(),
          icon: Icons.table_chart_rounded,
          label: 'Excel',
          isOutlined: true,
        ),
      );
    }

    return actions;
  }

  bool _canEditCurrentRevision(
    Map<String, dynamic> advance,
    Map<String, dynamic> item,
  ) {
    final status = (advance['status'] ?? '').toString();
    if (status == 'completed' ||
        status == 'approved' ||
        status == 'manager_approved') return false;

    final itemRevisionNo = item['revision_no'] ?? 0;
    if (status == 'draft' || status == 'rejected') {
      return itemRevisionNo == 0;
    }
    if (status == 'revision_draft' || status == 'revision_rejected') {
      return itemRevisionNo == (advance['active_revision_no'] ?? -1);
    }
    return false;
  }

  String _displayAdvanceTitle(Map<String, dynamic> advance) {
    final type = (advance['advance_type'] ?? 'single').toString().toLowerCase();
    final title = (advance['title'] ?? '').toString();

    if (type == 'single') {
      final items = List<Map<String, dynamic>>.from(
        advance['items'] as List? ?? [],
      );
      if (items.isNotEmpty) {
        // Sort by ID descending to get the latest item
        items.sort((a, b) => (b['id'] ?? 0).compareTo(a['id'] ?? 0));
        final latestItem = items.first;
        final desc = latestItem['description']?.toString() ?? '';
        return desc.isNotEmpty
            ? desc
            : (title.isNotEmpty ? title : 'Kasbon Mandiri');
      }
      return title.isNotEmpty ? title : 'Kasbon Mandiri';
    }
    return title.isEmpty ? 'Detail Kasbon' : title;
  }

  bool _hasUncheckedChecklist(Map<String, dynamic> item) {
    final checklist = _parseChecklist(item['notes']);
    return checklist.isNotEmpty &&
        checklist.any((entry) => entry['checked'] != true);
  }

  bool _canSubmitAdvanceItems(
      List<Map<String, dynamic>> items, Map<String, dynamic> adv) {
    if (items.isEmpty) return false;

    final itemsOk = !items.any(_hasUncheckedChecklist);
    final returnNotes = adv['return_notes'] ?? '';
    final returnChecklist = _parseChecklist(returnNotes);
    final returnOk = returnChecklist.isEmpty ||
        !returnChecklist.any((item) => item['checked'] != true);

    return itemsOk && returnOk;
  }

  bool _canApproveAdvanceItems(List<Map<String, dynamic>> items) {
    if (items.isEmpty) return false;
    return items.every(
      (item) =>
          (item['status'] ?? 'pending').toString().toLowerCase() == 'approved',
    );
  }

  List<Map<String, dynamic>> _workflowItems(
    Map<String, dynamic> advance,
    List<Map<String, dynamic>> items,
  ) {
    final status = (advance['status'] ?? '').toString().toLowerCase();
    final activeRevisionNo =
        int.tryParse('${advance['active_revision_no'] ?? ''}') ?? -1;

    if (status == 'revision_draft' ||
        status == 'revision_rejected' ||
        status == 'revision_submitted') {
      return items
          .where(
            (item) =>
                (int.tryParse('${item['revision_no'] ?? 0}') ?? 0) ==
                activeRevisionNo,
          )
          .toList();
    }

    return items
        .where(
          (item) => (int.tryParse('${item['revision_no'] ?? 0}') ?? 0) == 0,
        )
        .toList();
  }

  double _toDouble(dynamic value) {
    if (value is num) return value.toDouble();
    if (value == null) return 0.0;
    String str = value.toString().trim();
    if (str.isEmpty) return 0.0;

    // Handle Indonesian format: 1.234.567,89 -> 1234567.89
    // Or our new formatter output: 1.234.567.89 -> 1234567.89

    int dots = '.'.allMatches(str).length;
    int commas = ','.allMatches(str).length;

    if (dots > 1) {
      // Multiple dots. In our project, the last one is likely the decimal separator
      // added by our new formatter, OR they are all thousands separators.
      if (commas == 1) {
        str = str.replaceAll('.', '').replaceFirst(',', '.');
      } else {
        // Check if the last dot looks like a decimal (not followed by exactly 3 digits, or only 1-2 digits)
        int lastDotIdx = str.lastIndexOf('.');
        if (str.length - lastDotIdx != 4) {
          // Last dot is decimal
          String integerPart = str.substring(0, lastDotIdx).replaceAll('.', '');
          String decimalPart = str.substring(lastDotIdx + 1);
          str = "$integerPart.$decimalPart";
        } else {
          // All dots are thousands
          str = str.replaceAll('.', '');
        }
      }
    } else if (dots == 1 && commas == 1) {
      int dotIdx = str.indexOf('.');
      int commaIdx = str.indexOf(',');
      if (dotIdx < commaIdx) {
        str = str.replaceAll('.', '').replaceFirst(',', '.');
      } else {
        str = str.replaceAll(',', '');
      }
    } else if (commas > 1) {
      str = str.replaceAll(',', '');
    } else if (dots == 1) {
      // Single dot.
      int dotIdx = str.lastIndexOf('.');
      if (str.length - dotIdx == 4 && str.length > 4) {
        // Likely thousands, e.g., 1.234 or 10.234
        str = str.replaceAll('.', '');
      }
    } else if (commas == 1) {
      str = str.replaceFirst(',', '.');
    }

    return double.tryParse(str) ??
        double.tryParse(str.replaceAll(RegExp(r'[^\d.]'), '')) ??
        0.0;
  }

  Future<void> _approveAdvance() async {
    final currentAdvance = context.read<AdvanceProvider>().currentAdvance;
    if (currentAdvance == null) return;

    final confirmed = await AppDialogs.showConfirmDialog(
      context,
      title: 'Konfirmasi Approved Final',
      message:
          'Apakah Anda yakin ingin menyetujui final kasbon ini? Data akan dicatat ke Bank.',
      confirmLabel: 'Ya, Setujui',
      confirmColor: AppTheme.success,
      icon: Icons.check_circle_outline_rounded,
    );

    if (!confirmed) return;
    if (!mounted) return;

    setState(() => _isSaving = true);
    final prov = context.read<AdvanceProvider>();
    final success = await prov.finalApproveAdvance(widget.advanceId);

    if (!mounted) return;
    if (success && context.mounted) {
      await prov.loadAdvance(widget.advanceId); // ✅ Ensure UI refresh
      _showGlobalSnackBar(
        'Kasbon berhasil disetujui final ✓',
        AppTheme.success,
      );
    } else if (context.mounted) {
      _showGlobalSnackBar(
        prov.error ?? 'Gagal menyetujui final kasbon',
        AppTheme.danger,
      );
    }
    setState(() => _isSaving = false);
  }

                        final children = _asMapList(

                            ),

                          messenger.showSnackBar(
                            const SnackBar(
                              content: Text(
                                'Isi nominal dengan nilai yang benar.',

                              backgroundColor: AppTheme.danger,
                            ),
                          );
                          return;
                        }
                        final rateForValidation =
                            _toDouble(exchangeRateCtrl.text);
                        final totalIdrValidation = amount * rateForValidation;

                        if (totalIdrValidation.abs() < 100) {
                          messenger.showSnackBar(
                            const SnackBar(
                              content: Text(
                                'Nominal ekuivalen Rupiah harus lebih dari Rp 100',
                              ),
                              backgroundColor: AppTheme.danger,
                            ),
                          );
                          return;
                        }
                        // Apply negative sign if toggled
                        final finalAmount = isNegative ? -amount : amount;

                        // Lock mechanism - prevent double-tap
                        setState(() => _isSaving = true);

                        try {
                          final firstSubId = selectedSubCategoryIds.isNotEmpty
                              ? selectedSubCategoryIds.first
                              : null;
                          final finalCatId = firstSubId ?? selectedParentId!;
                          final subcatIds = selectedSubCategoryIds.toList();
                          final prov = context.read<AdvanceProvider>();

                          final existingPaths = selectedFilePaths
                              .where((p) => p.startsWith('receipts/'))
                              .toList();
                          final newPaths = selectedFilePaths
                              .where((p) => !p.startsWith('receipts/'))
                              .toList();

                          // For existing filenames, we need to match them with existing paths
                          final existingNames = <String>[];
                          for (var i = 0; i < selectedFilePaths.length; i++) {
                            if (selectedFilePaths[i].startsWith('receipts/')) {
                              if (i < selectedFileNames.length) {
                                existingNames.add(selectedFileNames[i]);
                              } else {
                                existingNames.add(
                                  selectedFilePaths[i].split('/').last,
                                );
                              }
                            }
                          }

                          bool success;
                          if (item == null) {
                            final items =
                                prov.currentAdvance?['items'] as List? ?? [];
                            final isFirstItem = items.isEmpty;

                            if (isFirstItem) {
                              success =
                                  await prov.saveFirstItemAndCommitAdvance(
                                advanceId: widget.advanceId,
                                categoryId: finalCatId,
                                categoryIds: subcatIds,
                                desc: descCtrl.text.trim(),
                                amount: finalAmount,
                                date: dateCtrl.text,
                                source: selectedSource,
                                currency: selectedCurrency,
                                currencyExchange: rateForValidation,
                                filePaths:
                                    newPaths.isNotEmpty ? newPaths : null,
                              );
                            } else {
                              success = await prov.addAdvanceItem(
                                widget.advanceId,
                                finalCatId,
                                descCtrl.text.trim(),
                                finalAmount,
                                categoryIds: subcatIds,
                                date: dateCtrl.text,
                                source: selectedSource,
                                currency: selectedCurrency,
                                currencyExchange: rateForValidation,
                                filePaths:
                                    newPaths.isNotEmpty ? newPaths : null,
                              );
                            }
                          } else {
                            success = await prov.updateAdvanceItem(
                              item['id'],
                              finalCatId,
                              descCtrl.text.trim(),
                              finalAmount,
                              categoryIds: subcatIds,
                              date: dateCtrl.text,
                              source: selectedSource,
                              currency: selectedCurrency,
                              currencyExchange: rateForValidation,
                              filePaths: newPaths.isNotEmpty ? newPaths : null,
                              existingFilePaths: existingPaths,
                              existingFileNames: existingNames,
                              status: item['status'],
                            );
                          }

                          if (ctx.mounted) Navigator.pop(ctx);
                          if (success && mounted) {
                            messenger.showSnackBar(
                              SnackBar(
                                content: Text(
                                  isEditing
                                      ? 'Item diupdate ✓'
                                      : 'Item ditambahkan ✓',
                                ),
                                backgroundColor: AppTheme.success,
                                duration: const Duration(seconds: 2),
                              ),
                            );
                            prov.loadAdvance(widget.advanceId);
                          }
                        } finally {
                          // Unlock mechanism
                          if (mounted) setState(() => _isSaving = false);
                        }
                      },
                child: _isSaving
                    ? const SizedBox(
                        width: 16,
                        height: 16,
                        child: CircularProgressIndicator(
                          strokeWidth: 2,
                          valueColor: AlwaysStoppedAnimation<Color>(
                            Colors.white,
                          ),
                        ),
                      )
                    : const Text('Simpan'),
              ),
            ],
          );
        },
      ),
    );

    // Delayed dispose to prevent "used after disposed" during closing animation
    Future.delayed(const Duration(milliseconds: 300), () {
      descCtrl.dispose();
      amountCtrl.dispose();
      dateCtrl.dispose();
      exchangeRateCtrl.dispose();
    });
  }

                          child: CircularProgressIndicator(
                            color: AppTheme.primary,
                          ),
                        )
                      : ResponsiveLayout.isMobile(context)
                          ? AppScrollbar(
                              controller: _mainScrollController,
                              thumbVisibility: true,
                              interactive: true,
                              child: SingleChildScrollView(
                                controller: _mainScrollController,
                                child: Column(

                                      canEditHeader,
                                      canSubmit,

                                      canStartRevision,
                                      canShowDownloadButtons,

                                  ],
                                ),
                              ),
                            )
                          : AppScrollbar(
                              controller: _mainScrollController,
                              thumbVisibility: true,
                              interactive: true,
                              child: SingleChildScrollView(
                                controller: _mainScrollController,
                                child: _buildContent(
                                    context, adv, auth, prov, items),

                  floatingActionButton: canAddItems

                          padding: const EdgeInsets.only(bottom: 16, right: 8),

                              adv['status'] == 'revision_draft' ||

              ), // End Expanded
            ], // End Row
          ), // End Row
        ), // End SelectionArea
      ), // End outer Scaffold
    ); // End PopScope
  }

  Widget _buildContent(
    BuildContext context,
    Map<String, dynamic> adv,
    AuthProvider auth,
    AdvanceProvider prov,
    List<Map<String, dynamic>> items,
  ) {
    final approved = adv['approved_amount'] ?? 0;
    final warnings = List<String>.from(adv['policy_warnings'] ?? []);
    final canEditItems =
        (auth.user?['id'] == adv['user_id'] || auth.isManager) &&
            [
              'draft',
              'rejected',
              'revision_draft',
              'revision_rejected',
            ].contains(adv['status']);

    final screenWidth = MediaQuery.of(context).size.width;
    final useCompact = screenWidth < 600 ||
        Theme.of(context).platform == TargetPlatform.android;

    return Padding(
      padding: EdgeInsets.all(useCompact ? 16 : 24),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          LayoutBuilder(
            builder: (context, constraints) {
              final isNarrow = constraints.maxWidth < 650;
              final currentTotal = items.fold(0.0, (sum, item) {
                final estimated = (item['estimated_amount'] ?? 0) as num;
                final kurs = (item['currency_exchange'] ?? 1.0) as num;
                final idrStr = item['idr_amount'];
                final idr =
                    (idrStr is num && idrStr > 0) ? idrStr : (estimated * kurs);
                return sum + idr.toDouble();
              });

              final cards = [
                SummaryCard(
                  icon: Icons.receipt_long_rounded,
                  label: 'Total',
                  value: 'Rp ${AppFormatters.formatCurrency(currentTotal)}',
                  color: AppTheme.primary,
                ),
                SummaryCard(
                  icon: Icons.verified_rounded,
                  label: 'Approved',
                  value: 'Rp ${AppFormatters.formatCurrency(approved)}',
                  color: AppTheme.success,
                ),
                SummaryCard(
                  icon: Icons.account_balance_wallet_rounded,
                  label: 'Reimbursement',
                  value: 'Rp ${AppFormatters.formatCurrency(approved)}',
                  color: AppTheme.warning,
                ),
                SummaryCard(
                  icon: Icons.list_alt_rounded,
                  label: 'Expense',
                  value: '${items.length} item',
                  color: AppTheme.accent,
                ),
              ];

              if (isNarrow) {
                return Column(
                  children: [
                    cards[0],
                    const SizedBox(height: 8),
                    cards[1],
                    const SizedBox(height: 8),
                    cards[2],
                    const SizedBox(height: 8),
                    cards[3],
                  ],
                );
              }

              return Row(
                children: [
                  Expanded(child: cards[0]),
                  const SizedBox(width: 12),
                  Expanded(child: cards[1]),
                  const SizedBox(width: 12),
                  Expanded(child: cards[2]),
                  const SizedBox(width: 12),
                  Expanded(child: cards[3]),
                ],
              );
            },
          ),
          const SizedBox(height: 12),
          Row(
            children: [
              Icon(
                Icons.person_outline,
                color: _secondaryText(context),
                size: 18,
              ),
              const SizedBox(width: 8),
              Text(
                'Dibuat oleh: ${adv['requester_name'] ?? '-'}',
                style: TextStyle(color: _secondaryText(context)),
              ),
              const SizedBox(width: 16),
              Icon(
                Icons.calendar_today_outlined,
                color: _secondaryText(context),
                size: 18,
              ),
              const SizedBox(width: 8),
              Text(
                adv['created_at'] != null
                    ? adv['created_at'].toString().substring(0, 10)
                    : '-',
                style: TextStyle(color: AppTheme.textSecondary),
              ),
              const SizedBox(width: 16),
              Icon(
                Icons.assignment_outlined,
                color: _secondaryText(context),
                size: 18,
              ),
              const SizedBox(width: 8),
              Text(
                'Laporan Tahunan: ${adv['report_year'] ?? '-'}',
                style: TextStyle(color: _secondaryText(context)),
              ),
            ],
          ),
          if (adv['notes'] != null && adv['notes'].toString().isNotEmpty) ...[
            const SizedBox(height: 16),
            Container(
              padding: const EdgeInsets.all(12),
              decoration: BoxDecoration(
                color: AppTheme.warning.withValues(alpha: 0.1),
                borderRadius: BorderRadius.circular(8),
                border: Border.all(
                  color: AppTheme.warning.withValues(alpha: 0.3),
                ),
              ),
              child: Row(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Icon(Icons.info_outline, color: AppTheme.warning, size: 20),
                  const SizedBox(width: 8),
                  Expanded(
                    child: Text(
                      adv['notes'],
                      style: TextStyle(color: _creamColor(context)),
                    ),
                  ),
                ],
              ),
            ),
          ],
          if (warnings.isNotEmpty) ...[
            const SizedBox(height: 16),
            ...warnings.map(_buildWarningBox),
          ],
          const SizedBox(height: 28),
          Row(
            children: [
              Text(
                'Rincian Item Kasbon',
                style: TextStyle(
                  fontSize: 18,
                  fontWeight: FontWeight.bold,
                  color: _titleColor(context),
                ),
              ),
              if (items.isNotEmpty) ...[
                const SizedBox(width: 12),
                Builder(builder: (context) {
                  final mainStatus =
                      (adv['status'] ?? 'draft').toString().toLowerCase();
                  final allApproved = items.every((item) {
                        final st = (item['status'] ?? 'pending')
                            .toString()
                            .toLowerCase();
                        return st == 'approved' ||
                            st == 'completed' ||
                            st == 'in_settlement';
                      }) ||
                      mainStatus == 'completed' ||
                      mainStatus == 'approved';

                  return Container(
                    padding:
                        const EdgeInsets.symmetric(horizontal: 10, vertical: 4),
                    decoration: BoxDecoration(
                      color: allApproved
                          ? AppTheme.success.withValues(alpha: 0.1)
                          : AppTheme.warning.withValues(alpha: 0.1),
                      borderRadius: BorderRadius.circular(20),
                      border: Border.all(
                          color:
                              allApproved ? AppTheme.success : AppTheme.warning,
                          width: 1),
                    ),
                    child: Text(
                      allApproved ? 'APPROVED' : 'PENDING',
                      style: TextStyle(
                        fontSize: 10,
                        fontWeight: FontWeight.bold,
                        color:
                            allApproved ? AppTheme.success : AppTheme.warning,
                      ),
                    ),
                  );
                }),
              ],
            ],
          ),
          const SizedBox(height: 16),
          if (items.isEmpty)
            _buildEmptyState(canEditItems)
          else
            _buildItemsTable(items, adv, auth),
          const SizedBox(height: 20),
          _buildAdvanceFundTransferSection(context, adv, auth),
          _buildAdvanceFundReturnSection(context, adv, auth),
          SizedBox(
            height: useCompact ? 80 : 100,
          ), // Padding extra agar tidak tertabrak FAB
        ],
      ),
    );
  }

  Widget _buildItemsTable(
    List<Map<String, dynamic>> items,
    Map<String, dynamic> adv,
    AuthProvider auth,
  ) {
    return Container(
      decoration: BoxDecoration(
        color: _cardColor(context),
        borderRadius: BorderRadius.circular(16),
        border: Border.all(color: _dividerColor(context)),
      ),
      child: ClipRRect(
        borderRadius: BorderRadius.circular(16),
        child: AppScrollbar(
          controller: _itemTableVerticalCtrl,
          thumbVisibility: true,
          trackVisibility: true,
          interactive: true,
          child: SingleChildScrollView(
            controller: _itemTableVerticalCtrl,
            child: AppScrollbar(
              controller: _itemTableHorizontalCtrl,
              thumbVisibility: true,
              trackVisibility: true,
              interactive: true,
              scrollDirection: Axis.horizontal,
              notificationPredicate: (notification) =>
                  notification.metrics.axis == Axis.horizontal,
              child: SingleChildScrollView(
                controller: _itemTableHorizontalCtrl,
                scrollDirection: Axis.horizontal,
                child: DataTable(
                  dataRowMaxHeight: double.infinity,
                  headingRowColor: WidgetStateProperty.all(
                    _surfaceColor(context),
                  ),
                  dataRowColor: WidgetStateProperty.resolveWith((states) {
                    if (states.contains(WidgetState.hovered)) {
                      return _isDark(context)
                          ? AppTheme.cardHover
                          : AppTheme.lightCardHover;
                    }
                    return _cardColor(context);
                  }),
                  columns: [
                    DataColumn(
                      label: Text(
                        'No',
                        style: TextStyle(
                          fontWeight: FontWeight.w600,
                          color: _creamColor(context),
                        ),
                      ),
                    ),
                    DataColumn(
                      label: Text(
                        'Subkategori',
                        style: TextStyle(
                          fontWeight: FontWeight.w600,
                          color: _creamColor(context),
                        ),
                      ),
                    ),
                    DataColumn(
                      label: Text(
                        'Deskripsi',
                        style: TextStyle(
                          fontWeight: FontWeight.w600,
                          color: _creamColor(context),
                        ),
                      ),
                    ),
                    DataColumn(
                      label: Text(
                        'Tanggal',
                        style: TextStyle(
                          fontWeight: FontWeight.w600,
                          color: _creamColor(context),
                        ),
                      ),
                    ),
                    DataColumn(
                      label: Text(
                        'Estimasi Biaya',
                        style: TextStyle(
                          fontWeight: FontWeight.w600,
                          color: _creamColor(context),
                        ),
                      ),
                      numeric: true,
                    ),
                    DataColumn(
                      label: Text(
                        'Evidence',
                        style: TextStyle(
                          fontWeight: FontWeight.w600,
                          color: _creamColor(context),
                        ),
                      ),
                    ),
                    DataColumn(
                      label: Text(
                        'Status',
                        style: TextStyle(
                          fontWeight: FontWeight.w600,
                          color: _creamColor(context),
                        ),
                      ),
                    ),
                    DataColumn(
                      label: Text(
                        'Aksi',
                        style: TextStyle(
                          fontWeight: FontWeight.w600,
                          color: _creamColor(context),
                        ),
                      ),
                    ),
                  ],
                  rows: items.asMap().entries.map((entry) {
                    final idx = entry.key;
                    final item = entry.value;
                    final itemId = item['id'] as int;
                    final itemStatus =
                        (item['status'] ?? 'pending').toString().toLowerCase();
                    final isItemApproved = itemStatus == 'approved';
                    final canEditRow = _canEditCurrentRevision(adv, item);
                    final isSubmittedState = adv['status'] == 'submitted' ||
                        adv['status'] == 'revision_submitted';
                    final showChecklist =
                        _isManager && isSubmittedState && !isItemApproved;

                    String subCat = '-';
                    final catName = item['category_name'] ?? '';
                    if (catName.contains(' > ')) {
                      final parts = catName.split(' > ');
                      subCat = parts.last;
                    } else {
                      subCat = catName;
                    }

                    final previewBtn = IconButton(
                      icon: const Icon(
                        Icons.info_outline_rounded,
                        color: AppTheme.primary,
                        size: 18,
                      ),
                      tooltip: 'Pratinjau Rincian',
                      onPressed: () => _showItemPreview(item),
                    );

                    return DataRow(
                      selected: showChecklist
                          ? _checkedItemIds.contains(itemId)
                          : _selectedItemIds.contains(itemId),
                      onSelectChanged: (selected) {
                        setState(() {
                          if (showChecklist) {
                            if (selected == true) {
                              _checkedItemIds.add(itemId);
                            } else {
                              _checkedItemIds.remove(itemId);
                            }
                          } else {
                            if (selected == true) {
                              _selectedItemIds.add(itemId);
                            } else {
                              _selectedItemIds.remove(itemId);
                            }
                          }
                        });
                      },
                      cells: [
                        DataCell(Text('${idx + 1}')),
                        DataCell(
                          ConstrainedBox(
                            constraints: const BoxConstraints(maxWidth: 160),
                            child: Text(subCat),
                          ),
                        ),
                        DataCell(
                          ConstrainedBox(
                            constraints: const BoxConstraints(maxWidth: 280),
                            child: Text(
                              item['description'] ?? '',
                            ),
                          ),
                        ),
                        DataCell(Text(item['date'] ?? '-')),
                        DataCell(
                          Builder(
                            builder: (context) {
                              final currency = (item['currency'] ?? 'IDR')
                                  .toString()
                                  .toUpperCase();
                              final estimated =
                                  (item['estimated_amount'] ?? 0) as num;
                              final kurs =
                                  (item['currency_exchange'] ?? 1.0) as num;
                              final idrStr = item['idr_amount'];
                              final idr = (idrStr is num && idrStr > 0)
                                  ? idrStr
                                  : (estimated * kurs);

                              if (currency == 'IDR') {
                                return Text(
                                  'Rp ${AppFormatters.formatCurrency(estimated)}',
                                );
                              }
                              final foreignFormat = NumberFormat(
                                '#,##0.##',
                                'en_US',
                              );
                              return Text(
                                '$currency ${foreignFormat.format(estimated)}\n(Rp ${AppFormatters.formatCurrency(idr)})',
                              );
                            },
                          ),
                        ),
                        DataCell(
                          item['evidence_path'] != null
                              ? IconButton(
                                  icon: Icon(
                                    Icons.image_rounded,
                                    color: AppTheme.accent,
                                    size: 20,
                                  ),
                                  tooltip: 'Lihat Bukti',
                                  onPressed: () => _showEvidence(
                                    item['evidence_path'],
                                    item['description'] ?? 'Evidence',
                                  ),
                                )
                              : const Text('-'),
                        ),
                        DataCell(
                          Builder(
                            builder: (context) {
                              final itemStatus = (item['status'] ?? 'pending')
                                  .toString()
                                  .toUpperCase();
                              Color statusColor = AppTheme.warning;

                              if (itemStatus == 'APPROVED') {
                                statusColor = AppTheme.success;
                              } else if (itemStatus == 'REJECTED') {
                                statusColor = AppTheme.danger;
                              }

                              final hasNotes = (itemStatus == 'REJECTED' ||
                                      itemStatus == 'PENDING') &&
                                  item['notes'] != null &&
                                  (item['notes'] as String).isNotEmpty;

                              List<dynamic> checklist = [];
                              if (hasNotes) {
                                try {
                                  checklist = _parseChecklist(item['notes']);
                                } catch (_) {}
                              }

                              final bool hasComments = checklist.isNotEmpty;

                              Widget badge = Container(
                                padding: const EdgeInsets.symmetric(
                                  horizontal: 10,
                                  vertical: 6,
                                ),
                                decoration: BoxDecoration(
                                  color: statusColor.withValues(alpha: 0.1),
                                  borderRadius: BorderRadius.circular(6),
                                  border: hasComments
                                      ? Border.all(
                                          color: statusColor.withValues(
                                            alpha: 0.3,
                                          ),
                                        )
                                      : null,
                                ),
                                child: Text(
                                  hasComments
                                      ? '$itemStatus (Comment ${checklist.where((it) => it['checked'] == true).length}/${checklist.length})'
                                      : itemStatus,
                                  style: TextStyle(
                                    fontSize: 10,
                                    fontWeight: FontWeight.w700,
                                    color: statusColor,
                                  ),
                                ),
                              );

                              if (hasComments) {
                                return InkWell(
                                  onTap: () => _showChecklistDialog(
                                    itemId,
                                    checklist,
                                    (adv['status'] ?? '').toString(),
                                  ),
                                  borderRadius: BorderRadius.circular(6),
                                  child: badge,
                                );
                              }

                              return badge;
                            },
                          ),
                        ),
                        DataCell(
                          Row(
                            mainAxisSize: MainAxisSize.min,
                            children: [
                              previewBtn,
                              if (canEditRow) ...[
                                IconButton(
                                  icon: const Icon(
                                    Icons.edit_rounded,
                                    size: 18,
                                    color: AppTheme.primary,
                                  ),
                                  tooltip: 'Edit Item',
                                  onPressed: () =>
                                      _showAddItemDialog(context, item),
                                ),
                                IconButton(
                                  icon: const Icon(
                                    Icons.delete_outline_rounded,
                                    size: 18,
                                    color: AppTheme.danger,
                                  ),
                                  tooltip: 'Hapus Item',
                                  onPressed: () => _deleteItem(itemId),
                                ),
                              ],
                              if (!canEditRow &&
                                  _isManager &&
                                  isSubmittedState) ...[
                                if (itemStatus == 'pending')
                                  IconButton(
                                    icon: const Icon(
                                      Icons.check_rounded,
                                      size: 18,
                                      color: AppTheme.success,
                                    ),
                                    tooltip: 'Approve Item',
                                    onPressed: () => _approveItem(itemId),
                                  ),
                                if (itemStatus == 'pending' ||
                                    itemStatus == 'approved')
                                  IconButton(
                                    icon: const Icon(
                                      Icons.close_rounded,
                                      size: 18,
                                      color: AppTheme.danger,
                                    ),
                                    tooltip: 'Reject Item',
                                    onPressed: () => _rejectItem(itemId),
                                  ),
                              ],
                            ],
                          ),
                        ),
                      ],
                    );
                  }).toList(),
                ),
              ),
            ),
          ),
        ),
      ),
    );
  }

  String _formatItemAmount(Map<String, dynamic> item) {
    final currency = (item['currency'] ?? 'IDR').toString().toUpperCase();
    final estimated = (item['estimated_amount'] ?? 0) as num;
    final kurs = (item['currency_exchange'] ?? 1.0) as num;
    final idrStr = item['idr_amount'];
    final idr = (idrStr is num && idrStr > 0) ? idrStr : (estimated * kurs);

    if (currency == 'IDR') {
      return 'Rp ${_currencyFormat.format(estimated)}';
    }
    final foreignFormat = NumberFormat('#,##0.##', 'en_US');
    return '$currency ${foreignFormat.format(estimated)} (Rp ${_currencyFormat.format(idr)})';
  }

  List<String> _getCommentsFromNotes(String notes) {
    if (notes.isEmpty) return [];

    final systemPrefixes = [
      'Disetujui oleh',
      'Imported from',
      'Subcategory:',
      'Approved by',
      'System:',
    ];

    try {
      final decoded = json.decode(notes);
      if (decoded is List) {
        final List<String> comments = [];
        for (var item in decoded) {
          final text = (item is Map ? item['text'] : item).toString();

          bool isSystem = false;
          for (var prefix in systemPrefixes) {
            if (text.trim().startsWith(prefix)) {
              isSystem = true;
              break;
            }
          }

          if (!isSystem && text.trim().isNotEmpty) {
            comments.add(text.trim());
          }
        }
        return comments;
      }
      return [];
    } catch (_) {
      bool isSystem = false;
      for (var prefix in systemPrefixes) {
        if (notes.trim().startsWith(prefix)) {
          isSystem = true;
          break;
        }
      }
      return isSystem || notes.trim().isEmpty ? [] : [notes.trim()];
    }
  }

  List<String> _parsePlainComments(dynamic raw) {
    if (raw == null) return [];
    final text = raw.toString().trim();
    if (text.isEmpty) return [];

    try {
      final decoded = jsonDecode(text);
      if (decoded is List) {
        return decoded
            .map((e) => e.toString().trim())
            .where((e) => e.isNotEmpty)
            .toList();
      }
    } catch (_) {}

    return text
        .split('\n')
        .map((e) => e.trim())
        .where((e) => e.isNotEmpty)
        .toList();
  }

          .map((c) => c.text.trim())
          .where((t) => t.isNotEmpty)
          .map((t) => {'text': t, 'checked': false})

                backgroundColor: AppTheme.danger,
              ),
            );
          }
        }
      }
    }

    // Always dispose after everything is done
    Future.delayed(const Duration(milliseconds: 300), () {
      for (var c in controllers) {
        try {
          c.dispose();
        } catch (_) {}
      }
    });
  }

    BuildContext ctx,
    StateSetter setDialogState,
    Function(Map<String, dynamic>) onCreated, {
    int? parentId,
  }) async {
    final nameCtrl = TextEditingController();
    final prov = context.read<SettlementProvider>();

    await showDialog(
      context: ctx,
      builder: (innerCtx) => AlertDialog(
        backgroundColor: _cardColor(innerCtx),

        title: Text(
          parentId != null ? 'Tambah Sub Kategori' : 'Tambah Kategori Utama',
          style: TextStyle(color: _creamColor(innerCtx)),
        ),
        content: SizedBox(
          width: 350,
          child: TextField(
            controller: nameCtrl,
            decoration: InputDecoration(
              labelText:
                  parentId != null ? 'Nama Sub Kategori' : 'Nama Kategori',
            ),
            style: TextStyle(color: _primaryText(ctx)),
            autofocus: true,
          ),
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(innerCtx),
            child: const Text('Batal'),
          ),
          ElevatedButton(
            onPressed: () async {

              final success = await prov.createCategory(
                nameCtrl.text.trim(),
                parentId: parentId,
              );
              if (innerCtx.mounted) Navigator.pop(innerCtx);
              if (success && mounted) {
                // cari kategori baru untuk callback
                final all = prov.categories;
                Map<String, dynamic> found = {};

                // cari kategori yang baru dibuat
                for (final p in all) {
                  if (parentId == null && p['name'] == nameCtrl.text.trim()) {
                    found = p;
                    break;
                  }
                  if (parentId != null && p['id'] == parentId) {
                    final children = (p['children'] as List?) ?? [];
                    for (final c in children) {
                      if (c['name'] == nameCtrl.text.trim()) {
                        found = c;
                        break;
                      }
                    }
                  }
                }

                onCreated(found);

                if (context.mounted) {
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(
                      content: Text(
                        'Kategori ditambahkan (menunggu approval manager) ✓',
                      ),
                      backgroundColor: AppTheme.success,
                    ),
                  );
                }
              }
            },
            child: const Text('Tambah'),
          ),
        ],
      ),
    );

    // Delayed dispose
    Future.delayed(const Duration(milliseconds: 300), () {
      nameCtrl.dispose();
    });
  }

  List<Map<String, dynamic>> _parseChecklist(dynamic checklist) {
    List<Map<String, dynamic>> local = [];
    try {
      dynamic parsed = checklist is String ? jsonDecode(checklist) : checklist;
      if (parsed is List) {
        for (final e in parsed) {
          String text = '';
          bool checked = false;

          if (e is Map) {
            text = (e['text'] ?? '').toString();
            checked = e['checked'] == true;
          } else if (e is String) {
            text = e;
          }

          if (text.isNotEmpty) {
            local.add({'text': text, 'checked': checked});
          }
        }
      }
    } catch (_) {}
    return local;
  }

                style: TextStyle(
                  color: _secondaryText(context),
                  decoration: TextDecoration.lineThrough,
                ),
              ),
            )
          : Text(
              item['text'] ?? '',
              style: TextStyle(color: _creamColor(context)),
            ),
      activeColor: AppTheme.success,
      checkColor: Colors.white,
      controlAffinity: ListTileControlAffinity.leading,
      contentPadding: EdgeInsets.zero,
    );
  }

  Widget _buildSaveChecklistButton(
    BuildContext dialogCtx,
    int itemId,
    List<Map<String, dynamic>> list,
    String status,
  ) {
    return ElevatedButton(
      style: ElevatedButton.styleFrom(
        backgroundColor: AppTheme.primary,
        foregroundColor: Colors.white,
      ),
      onPressed: () async {
        bool allFinished = list.every((it) => it['checked'] == true);
        final newStatus = allFinished ? 'pending' : 'rejected';

        final prov = context.read<AdvanceProvider>();
        final success = await prov.updateItemPartial(itemId, {
          'notes': jsonEncode(list),
          'status': newStatus,
        });

        if (success) {
          prov.loadAdvance(widget.advanceId);
          if (dialogCtx.mounted) Navigator.pop(dialogCtx);
          _showGlobalSnackBar(
            'Checklist berhasil disimpan ✓',
            AppTheme.success,
          );
        } else {
          if (dialogCtx.mounted) Navigator.pop(dialogCtx);
          _showGlobalSnackBar('Gagal menyimpan checklist', AppTheme.danger);
        }
      },
      child: const Text('Simpan'),
    );
  }

  Widget _buildFileList(
    List<String> paths,
    List<String>? filenames,
    bool isLocal,
    Function(int) onRemove,
    BuildContext ctx,
    StateSetter setDialogState,
  ) {
    if (paths.isEmpty) return const SizedBox.shrink();
    return Container(
      margin: const EdgeInsets.only(top: 8),
      padding: const EdgeInsets.all(8),
      decoration: BoxDecoration(
        color: Colors.black.withValues(alpha: 0.05),
        borderRadius: BorderRadius.circular(12),
        border: Border.all(color: _dividerColor(ctx)),
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            isLocal
                ? 'Terpilih (${paths.length}):'
                : 'File Lama (${paths.length}):',
            style: TextStyle(
              color: AppTheme.textSecondary,
              fontSize: 10,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 4),
          ...List.generate(paths.length, (index) {
            final name = (filenames != null && filenames.length > index)
                ? filenames[index]
                : paths[index].split('/').last;
            return Padding(
              padding: const EdgeInsets.symmetric(vertical: 2),
              child: Row(
                children: [
                  Icon(
                    isLocal ? Icons.check_circle : Icons.cloud_done,
                    color: isLocal ? AppTheme.success : AppTheme.primary,
                    size: 14,
                  ),
                  const SizedBox(width: 6),
                  Expanded(
                    child: Text(
                      name,
                      style: TextStyle(
                        color: isLocal ? AppTheme.success : AppTheme.cream,
                        fontSize: 11,
                      ),
                      maxLines: 1,
                      overflow: TextOverflow.ellipsis,
                    ),
                  ),
                  GestureDetector(
                    onTap: () {
                      setDialogState(() {
                        onRemove(index);
                      });
                    },
                    child: const Icon(
                      Icons.cancel,
                      size: 16,
                      color: AppTheme.danger,
                    ),
                  ),
                ],
              ),
            );
          }),
        ],
      ),
    );
  }

                        localReturnFilePaths.map((p) => File(p)).toList(),
                    existingFilePaths: localExistingPaths,
                    existingFileNames: localExistingNames,
                  );
                  setDialogState(() => _isSaving = false);

                  if (success && mounted) {
                    setState(() {
                      _returnFilePaths.clear();
                      _existingReturnPaths.clear();
                      _existingReturnFilenames.clear();
                    });
                    if (ctx.mounted) Navigator.pop(ctx);
                    _showGlobalSnackBar(
                        'Data pengembalian disimpan', AppTheme.success);
                  } else if (!success && context.mounted) {
                    ScaffoldMessenger.of(context).showSnackBar(
                      SnackBar(
                        content: Text(prov.error ?? 'Gagal menyimpan data'),
                        backgroundColor: AppTheme.danger,
                      ),
                    );
                  }
                },
                child: const Text('Simpan'),
              ),
            ],
          );
        },
      ),
    );
  }
}
